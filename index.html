def get_main_keyboard():
    markup = types.ReplyKeyboardMarkup(resize_keyboard=True, row_width=2)
    btn_theory = types.KeyboardButton('📖 Теория (Python)')
    btn_practice = types.KeyboardButton('📝 Практика (Тесты)')
    btn_math = types.KeyboardButton('🔢 Математика и ИТ')
    markup.add(btn_theory, btn_practice, btn_math)
    return markup

# --- КНОПКА "НАЗАД" ---
def get_back_keyboard():

@bot.message_handler(commands=['start'])
def start_message(message):
    student_name = message.from_user.first_name if message.from_user.first_name else "ученик"
    welcome_text = (
        f"Привет, {student_name}! 👋\n\n"
        "🤖 Я твой интерактивный помощник по **Информатике и программированию на Python**!\n\n"
        "Здесь ты сможешь легко разобрать сложные темы, потренироваться перед контрольными "

    bot.send_message(message.chat.id, welcome_text, reply_markup=get_main_keyboard(), parse_mode="Markdown")
